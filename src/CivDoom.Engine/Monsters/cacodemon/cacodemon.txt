speed          = 1.0   # walking speed (wall heights per second; you walk at 2.2)
size           = 0.5   # height of the [idle] picture (1.0 = a full wall)
float height   = 0.3   # hover above the floor (0 = walks on the floor)
damage         = 10-20   # fireball damage, a random number in this range
