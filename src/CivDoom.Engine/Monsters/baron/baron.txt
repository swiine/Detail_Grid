# ==================================================================
#  CivDOOM BOSS design file: BARON OF HELL
#
#  Bosses guard the finish line: "boss = yes" means it never turns up as
#  a random monster; each level puts its bosses in their own arenas and
#  the gate to the finish opens when they're beaten.
#
#  Edit in Notepad, save, press F5 in the game. Lines starting with #
#  are comments. Want a picture instead? Put baron.png or baron.jpg
#  in this folder (extra frames: baron_walk, baron_attack, baron_dead).
#
#  New boss: copy this whole folder and rename the folder and the .txt
#  inside (e.g. icon\icon.txt). Keep "boss = yes".
# ==================================================================

name = Baron of Hell
boss = yes

# ---- Stats -------------------------------------------------------
health         = 600   # hit points (a rocket does 60-100)
speed          = 1.4   # walking speed (you walk at 2.2)
size           = 1.2   # height of the [idle] picture (1.0 = a full wall)
float height   = 0   # hover above the floor (0 = walks on the floor)
damage         = 10-18   # damage per shot
shots          = 2   # shots per attack, fanned out across "shot spread"
shot spread    = 10   # degrees
shot color     = 40FF40   # colour of its shots
fireball speed = 5   # how fast its shots fly
attack delay   = 1.3   # seconds between attacks
spawn weight   = 1   # how often this boss is picked compared with other bosses
