# ==================================================================
#  CivDOOM BOSS design file: CYBERDEMON
#
#  Bosses guard the finish line: "boss = yes" means it never turns up as
#  a random monster; each level puts its bosses in their own arenas and
#  the gate to the finish opens when they're beaten.
#
#  Edit in Notepad, save, press F5 in the game. Lines starting with #
#  are comments. Want a picture instead? Put cyberdemon.png or cyberdemon.jpg
#  in this folder (extra frames: cyberdemon_walk, cyberdemon_attack, cyberdemon_dead).
#
#  New boss: copy this whole folder and rename the folder and the .txt
#  inside (e.g. icon\icon.txt). Keep "boss = yes".
# ==================================================================

name = Cyberdemon
boss = yes

# ---- Stats -------------------------------------------------------
health         = 900   # hit points (a rocket does 60-100)
speed          = 0.9   # walking speed (you walk at 2.2)
size           = 1.4   # height of the [idle] picture (1.0 = a full wall)
float height   = 0   # hover above the floor (0 = walks on the floor)
damage         = 25-40   # damage per shot
shots          = 1   # shots per attack, fanned out across "shot spread"
shot spread    = 0   # degrees
shot color     = FFB040   # colour of its shots
fireball speed = 5.5   # how fast its shots fly
attack delay   = 2.0   # seconds between attacks
spawn weight   = 1   # how often this boss is picked compared with other bosses
