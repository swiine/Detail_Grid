# ==================================================================
#  CivDOOM BOSS design file: SPIDER MASTERMIND
#
#  Bosses guard the finish line: "boss = yes" means it never turns up as
#  a random monster; each level puts its bosses in their own arenas and
#  the gate to the finish opens when they're beaten.
#
#  Edit in Notepad, save, press F5 in the game. Lines starting with #
#  are comments. Want a picture instead? Put mastermind.png or mastermind.jpg
#  in this folder (extra frames: mastermind_walk, mastermind_attack, mastermind_dead).
#
#  New boss: copy this whole folder and rename the folder and the .txt
#  inside (e.g. icon\icon.txt). Keep "boss = yes".
# ==================================================================

name = Spider Mastermind
boss = yes

# ---- Stats -------------------------------------------------------
health         = 800   # hit points (a rocket does 60-100)
speed          = 1.0   # walking speed (you walk at 2.2)
size           = 1.0   # height of the [idle] picture (1.0 = a full wall)
float height   = 0   # hover above the floor (0 = walks on the floor)
damage         = 4-8   # damage per shot
shots          = 5   # shots per attack, fanned out across "shot spread"
shot spread    = 25   # degrees
shot color     = FFFF80   # colour of its shots
fireball speed = 8   # how fast its shots fly
attack delay   = 0.9   # seconds between attacks
spawn weight   = 1   # how often this boss is picked compared with other bosses
