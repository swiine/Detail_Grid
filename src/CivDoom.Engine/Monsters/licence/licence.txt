name = Licence Expired
boss = yes

# ---- Stats -------------------------------------------------------
health         = 500   # hit points (a pistol shot does 12-25, a rocket 60-100)
speed          = 2.0   # walking speed (wall heights per second; you walk at 2.2)
size           = 1.25   # height of the [idle] picture (1.0 = a full wall)
float height   = 0   # hover above the floor (0 = walks on the floor)
damage         = 6-12   # damage per shot, a random number in this range
shots          = 1   # shots per attack, fanned out across "shot spread"
shot spread    = 0   # degrees
shot color     = FFD020   # colour of its shots
fireball speed = 9   # how fast its shots fly
attack delay   = 0.7   # seconds between attacks
spawn weight   = 1   # how common it is (bosses: how often picked)
glitch         = 0.3   # 0-1: how much it tears and flickers on screen

# ---- Fighting style ----------------------------------------------
behavior        = rusher   # skirmisher (keeps distance, circles), rusher (charges in),
                         # sniper (hangs back, leads shots), tank (never backs off)
preferred range = 0.5   # distance it likes to fight from (wall heights)
strafe          = 0.7   # 0-1: how much it moves sideways while fighting
aim lead        = 0.5   # 0-1: how far ahead of you it aims when you're moving
dodge           = 0.5   # 0-1: chance to side-step when you aim at it
melee damage    = 12-20   # close-range hit (0 = shoots only)

# ---- Colours -----------------------------------------------------
# One letter = one colour (hex RRGGBB). "." is always see-through.
[colors]
S = B8B8C0   shackle
Y = E8B820   padlock
y = A07810   padlock shadow
k = 202020   keyhole
R = FF2020   keyhole (attacking)
