name = Infinite Regen

speed          = 1.0   # walking speed (wall heights per second; you walk at 2.2)
size           = 1.1   # height of the [idle] picture (1.0 = a full wall)
float height   = 0   # hover above the floor (0 = walks on the floor)
damage         = 4-8   # damage per shot, a random number in this range
shots          = 5   # shots per attack, fanned out across "shot spread"
shot spread    = 25   # degrees
shot color     = FFD24A   # colour of its shots
fireball speed = 8   # how fast its shots fly
attack delay   = 0.9   # seconds between attacks
spawn weight   = 1   # how common it is (bosses: how often picked)
glitch         = 0.3   # 0-1: how much it tears and flickers on screen
